cmake_minimum_required(VERSION 3.20)
project(moLinkSessionService LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT _WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Windows")
  message(FATAL_ERROR "molink-service only builds for Windows")
endif()
add_executable(molink-service main.cpp)
target_link_libraries(molink-service PRIVATE advapi32 wtsapi32 user32)
if(MINGW)
  target_link_options(molink-service PRIVATE -municode)
endif()
install(TARGETS molink-service RUNTIME DESTINATION .)
